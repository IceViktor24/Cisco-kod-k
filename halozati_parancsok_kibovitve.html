Cisco IOS Enterprise Cheat Sheet

Interaktív hálózati parancssor-gyűjtemény vizsgákhoz és konfigurációhoz
Nincs a keresésnek megfelelő parancs.

SSH & Távmenedzsment (VTY)

# Alapfeltételek beállítása (Hostnév és Domain)
hostname R1
ip domain-name cisco.local
# RSA kulcspár generálása az SSH-hoz (min. 1024 bit ajánlott)
crypto key generate rsa
# Helyi felhasználó létrehozása titkosított jelszóval
username admin privilege 15 secret SzuperTitkosJelszo
# VTY (virtuális) vonalak megnyitása és SSH kényszerítése
line vty 0 4
 transport input ssh
 login local
 exit
# SSH verzió kényszerítése és állapot ellenőrzése
ip ssh version 2
show ip ssh
show ssh

EtherChannel & Link Aggregation (LACP)

# Interfészek kijelölése és LACP alapú csatornába rendezése
interface range interface-id
 channel-group 1 mode active
# A létrejött logikai Port-Channel interfész konfigurálása Trunk-ként
interface port-channel 1
 switchport mode trunk
 switchport trunk allowed vlan vlan-ids
# EtherChannel állapotának és a fizikai portok státuszának ellenőrzése
show etherchannel summary
show interfaces port-channel 1

Spanning Tree Protocol (STP)

# Switch kényszerítése Elsődleges / Másodlagos Root Bridge-nek
spanning-tree vlan vlan-id root primary
spanning-tree vlan vlan-id root secondary
# Root Bridge beállítása prioritási érték manuális módosításával
spanning-tree vlan vlan-id priority priority-value
# Gyorsított portváltás és BPDU védelem beállítása végponti portokon
interface interface-id
 spanning-tree portfast
 spanning-tree bpduguard enable
# STP topológia, Root ID és port-státuszok (Designated, Root, Altn) lekérése
show spanning-tree
show spanning-tree vlan vlan-id

Switchport Security (Port Biztonság)

# Port Security engedélyezése az access porton
interface interface-id
 switchport mode access
 switchport port-security
# Maximális engedélyezett MAC címek száma és dinamikus tanulás (sticky)
 switchport port-security maximum 2
 switchport port-security mac-address sticky
# Biztonsági szabálysértés módja (lekapcsolás)
 switchport port-security violation shutdown
# Portbiztonsági státuszok és szabályszegési számlálók ellenőrzése
show port-security
show port-security interface interface-id

VLAN-ok & Inter-VLAN Routing (ROAS)

# VLAN létrehozása és elnevezése
vlan vlan-id
 name vlan-name
# Router alinterfész konfigurálása és 802.1Q encapsulation (Trunk)
interface interface-id.subinterface-number
 encapsulation dot1Q vlan-id
 ip address ip-address subnet-mask
# VLAN adatbázis és Trunk linkek státuszának ellenőrzése
show vlan brief
show interfaces trunk

OSPF Dinamikus Irányítás

# OSPF folyamat indítása és Router ID manuális rögzítése
router ospf process-id
 router-id router-id-value
# Hálózatok hirdetése Wildcard maszkkal és Terület (Area) kiosztással
 network ip-address wildcard-mask area area-id
# Interfész passzívvá tétele (nem küld OSPF üzeneteket a végpontok felé)
 passive-interface interface-id
# OSPF szomszédok kapcsolatainak és az útválasztási táblának a megtekintése
show ip ospf neighbor
show ip route ospf

Access Control Lists (ACL - Hozzáférési Listák)

# Szabványos (Standard) számozott ACL létrehozása (csak forrás IP szűrés)
access-list 10 permit ip-address wildcard-mask
access-list 10 deny any
# Kiterjesztett (Extended) nevesített ACL (Forrás, Cél IP és specifikus portok)
ip access-list extended WEB_ONLY
 permit tcp source-ip wildcard destination-ip wildcard eq 80
 permit tcp source-ip wildcard destination-ip wildcard eq 443
 deny ip any any
# ACL alkalmazása egy interfész bejövő (in) vagy kimenő (out) forgalmára
interface interface-id
 ip access-group {10 | WEB_ONLY} {in | out}
# Aktív hozzáférési listák és a szabályokra illeszkedő csomagok száma
show access-lists

NAT & PAT (Címfordítás)

# Statikus NAT konfigurálása (Belső szerver fix publikus IP címre fordítása)
ip nat inside source static local-ip global-ip
# Dinamikus PAT (Port Address Translation) belső hálózat kiengedése egy külső interfészen
access-list 1 permit 192.168.1.0 0.0.0.255
ip nat inside source list 1 interface interface-id overload
# Belső (inside) és Külső (outside) interfészek kijelölése
interface interface-id-inside
 ip nat inside
interface interface-id-outside
 ip nat outside
# Aktív címfordítási tábla és statisztikák nyomon követése
show ip nat translations
show ip nat statistics

EtherChannel – LACP, PAgP és statikus

# LACP: active/active vagy active/passive párosítás
interface range gigabitEthernet 0/1 - 2
 switchport mode trunk
 channel-group 1 mode active
 no shutdown
exit
interface port-channel 1
 switchport mode trunk
# PAgP: desirable/auto vagy desirable/desirable (Cisco)
interface range gigabitEthernet 0/3 - 4
 channel-group 2 mode desirable
# Statikus EtherChannel (mindkét oldalon on)
interface range gigabitEthernet 0/5 - 6
 channel-group 3 mode on
# Ellenőrzés és hibakeresés
show etherchannel summary
show etherchannel detail
show lacp neighbor
show pagp neighbor
show interfaces port-channel 1
show interfaces trunk

STP – részletes konfiguráció

# PVST+ és root bridge beállítás
spanning-tree mode pvst
spanning-tree vlan 10 root primary
spanning-tree vlan 20 root secondary
spanning-tree vlan 10 priority 4096
# PortFast és BPDU Guard kizárólag végponti access porton
interface fastEthernet 0/10
 switchport mode access
 spanning-tree portfast
 spanning-tree bpduguard enable
# Root Guard: nem enged új root bridge-et a kijelölt port felől
interface gigabitEthernet 0/3
 spanning-tree guard root
# STP állapot és hibakeresés
show spanning-tree
show spanning-tree vlan 10
show spanning-tree root
show spanning-tree interface gigabitEthernet 0/1 detail
show spanning-tree summary

RSTP – Rapid PVST+

# Rapid PVST+ bekapcsolása (Cisco switch)
configure terminal
spanning-tree mode rapid-pvst
spanning-tree vlan 10 root primary
spanning-tree vlan 20 root secondary
# Edge port gyors továbbítása és BPDU védelem
interface fastEthernet 0/10
 switchport mode access
 spanning-tree portfast
 spanning-tree bpduguard enable
# RSTP ellenőrzés: Root, Designated, Alternate portok
show spanning-tree summary
show spanning-tree vlan 10
show spanning-tree interface gigabitEthernet 0/1 detail
show spanning-tree root
# Rapid PVST+ helyett klasszikus PVST+ mód
spanning-tree mode pvst

Subinterface – Router-on-a-Stick

# Fizikai router interfész bekapcsolása
interface gigabitEthernet 0/0
 no shutdown
# VLAN 10 és VLAN 20 alinterfész / gateway
interface gigabitEthernet 0/0.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
exit
interface gigabitEthernet 0/0.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
# Natív VLAN példa (csak ha a trunk másik oldalán is egyezik)
interface gigabitEthernet 0/0.99
 encapsulation dot1Q 99 native
 ip address 192.168.99.1 255.255.255.0
# Kapcsolódó switch port trunk módba állítása
interface gigabitEthernet 0/1
 switchport mode trunk
 switchport trunk allowed vlan 10,20,99
# Ellenőrzés
show ip interface brief
show running-config interface gigabitEthernet 0/0.10
show ip route
show interfaces trunk
